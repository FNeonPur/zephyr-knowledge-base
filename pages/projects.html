实战项目

从零到一，用 Zephyr RTOS 构建完整的物联网产品 —— 传感器节点、MQTT 网关、可穿戴设备、OTA 升级与自定义板移植

智能传感器节点

这是最经典的 Zephyr 入门实战项目。我们将使用 nRF52840 开发板搭配 BME280 温湿度气压传感器，通过 I2C 总线读取环境数据，然后通过 BLE 广播出去。设备在两次采样之间进入低功耗睡眠模式，实现超长续航。

项目架构

整个项目分为三个核心模块：传感器采集、BLE 广播和低功耗管理。主循环每次唤醒后依次执行：读传感器 → 组装广播数据 → BLE 广播 → 重新进入睡眠。

设备树覆盖层

BME280 挂在 I2C0 总线上，地址为 0x76。我们需要通过设备树覆盖层告诉 Zephyr 这个传感器的存在和连接方式。

dts 
/* boards/nrf52840dk_nrf52840.overlay */
&i2c0 {
    compatible = "nordic,nrf-twi";
    status = "okay";

    bme280: bme280@76 {
        compatible = "bosch,bme280";
        reg = <0x76>;
        label = "BME280";
    };
};

主程序

C 
#include <zephyr/kernel.h>
#include <zephyr/device.h>
#include <zephyr/drivers/sensor.h>
#include <zephyr/bluetooth/bluetooth.h>
#include <zephyr/bluetooth/conn.h>
#include <zephyr/pm/pm.h>
#include <zephyr/pm/device.h>
#include <stdio.h>

#define SLEEP_MS    5000
#define ADV_INTERVAL 100

static const struct bt_data ad[] = {
    BT_DATA_BYTES(BT_DATA_FLAGS,
                  (BT_LE_AD_GENERAL | BT_LE_AD_NO_BREDR)),
    BT_DATA_BYTES(BT_DATA_MANUFACTURER_DATA,
                  0x59, 0x00,   /* Nordic Semiconductor 公司 ID */
                  0x00, 0x00, 0x00,  /* 温度占位 */
                  0x00, 0x00,        /* 湿度占位 */
                  0x00, 0x00),       /* 气压占位 */
};

static void update_adv_data(double temp, double hum, double press)
{
    /* 直接修改 ad[1] 中 manufacturer data 的传感器字段 */
    uint8_t *data = (uint8_t *)ad[1].data + 2;

    int16_t t = (int16_t)(temp * 100);
    uint16_t h = (uint16_t)(hum * 100);
    uint16_t p = (uint16_t)(press / 10);

    data[0] = t & 0xFF;  data[1] = (t >> 8) & 0xFF;
    data[2] = h & 0xFF;  data[3] = (h >> 8) & 0xFF;
    data[4] = p & 0xFF;  data[5] = (p >> 8) & 0xFF;
}

int main(void)
{
    const struct device *bme = DEVICE_DT_GET_ANY(bosch_bme280);

    if (!device_is_ready(bme)) {
        printk("BME280 未就绪！\n");
        return -1;
    }

    bt_enable(NULL);
    bt_le_adv_start(BT_LE_ADV_PARAM(BT_LE_ADV_OPT_USE_IDENTITY,
                           BT_GAP_ADV_FAST_INT_MIN_2,
                           BT_GAP_ADV_FAST_INT_MAX_2, NULL),
                     ad, ARRAY_SIZE(ad), NULL, 0);

    printk("传感器节点启动完成\n");

    while (1) {
        struct sensor_value sv_temp, sv_hum, sv_press;

        sensor_sample_fetch(bme);
        sensor_channel_get(bme, SENSOR_CHAN_AMBIENT_TEMP, &sv_temp);
        sensor_channel_get(bme, SENSOR_CHAN_HUMIDITY, &sv_hum);
        sensor_channel_get(bme, SENSOR_CHAN_PRESS, &sv_press);

        double temp  = sensor_value_to_double(&sv_temp);
        double hum   = sensor_value_to_double(&sv_hum);
        double press = sensor_value_to_double(&sv_press);

        printk("T: %.2f°C  H: %.2f%%  P: %.2f hPa\n",
               temp, hum, press);

        update_adv_data(temp, hum, press);
        bt_le_adv_update_data(ad, ARRAY_SIZE(ad), NULL, 0);

        k_msleep(SLEEP_MS);
    }

    return 0;
}

小贴士：BLE 广播数据最大 31 字节，我们的 manufacturer data 段只用了 8 字节（2 字节公司 ID + 6 字节传感器数据），非常紧凑。如果需要广播更多数据，可以考虑使用 BLE Extended Advertising（扩展广播），最大支持 1650 字节。

低功耗关键：在 nRF52840 上，k_msleep() 会让 CPU 进入空闲睡眠。配合 CONFIG_PM=y，电源管理子系统会自动将设备推入更深的低功耗状态（如 System OFF），大幅降低功耗。实测睡眠电流可低至 1.5μA。

MQTT 物联网网关

在物联网系统中，传感器节点负责采集数据，而网关负责把数据转发到云端。本项目使用 ESP32 开发板，通过 WiFi 连接网络，使用 MQTT 协议与云平台通信——订阅控制主题接收指令，发布传感器主题上报数据。

项目架构

网关的核心工作流：WiFi 连接 → MQTT 连接 → 订阅控制主题 → 定时发布传感器数据 → 处理下行命令。我们使用 Zephyr 的网络管理接口来管理 WiFi 连接，使用 MQTT 客户端库来处理消息收发。

主程序

C 
#include <zephyr/kernel.h>
#include <zephyr/net/wifi_mgmt.h>
#include <zephyr/net/mqtt.h>
#include <zephyr/net/socket.h>
#include <zephyr/net/net_mgmt.h>
#include <stdio.h>

#define WIFI_SSID      "MyHomeWiFi"
#define WIFI_PSK       "my_password"
#define MQTT_BROKER    "192.168.1.100"
#define MQTT_PORT      1883
#define PUB_TOPIC      "sensor/data"
#define SUB_TOPIC      "sensor/cmd"

static struct mqtt_client client;
static uint8_t rx_buf[256];
static uint8_t tx_buf[256];
static struct sockaddr_in broker;

static void mqtt_evt_handler(struct mqtt_client *c,
                              const struct mqtt_evt *evt)
{
    switch (evt->type) {
    case MQTT_EVT_CONNACK:
        printk("MQTT 已连接\n");
        break;
    case MQTT_EVT_PUBLISH: {
        const struct mqtt_publish_param *p = &evt->param.publish;
        uint8_t payload[128];
        int len = mqtt_publish_payload_get(
                      c, p, payload, sizeof(payload) - 1);
        payload[len] = '\0';
        printk("收到命令: %s\n", payload);
        break;
    }
    default:
        break;
    }
}

static void mqtt_setup(void)
{
    mqtt_client_init(&client);

    broker.sin_family = AF_INET;
    broker.sin_port = htons(MQTT_PORT);
    net_addr_pton(AF_INET, MQTT_BROKER, &broker.sin_addr);

    struct mqtt_utf8 username = MQTT_UTF8_LITERAL("zephyr-gw");
    struct mqtt_utf8 password = MQTT_UTF8_LITERAL("gw-pass");

    client.broker = (struct sockaddr *)&broker;
    client.evt_cb = mqtt_evt_handler;
    client.client_id.utf8 = (uint8_t *)"zephyr-gw-01";
    client.client_id.size = sizeof("zephyr-gw-01") - 1;
    client.password = &password;
    client.user_name = &username;
    client.rx_buf = rx_buf;
    client.rx_buf_size = sizeof(rx_buf);
    client.tx_buf = tx_buf;
    client.tx_buf_size = sizeof(tx_buf);

    mqtt_connect(&client);
}

static void mqtt_subscribe_cmd(void)
{
    struct mqtt_topic sub_topic = {
        .topic.utf8 = (uint8_t *)SUB_TOPIC,
        .topic.size = strlen(SUB_TOPIC),
        .qos = MQTT_QOS_1_AT_LEAST_ONCE,
    };
    struct mqtt_subscription_list sub = {
        .list = &sub_topic,
        .list_count = 1,
    };
    mqtt_subscribe(&client, &sub);
}

static void publish_sensor_data(void)
{
    char payload[128];
    int64_t ts = k_uptime_get();

    snprintf(payload, sizeof(payload),
             "{\"ts\":%lld,\"temp\":23.5,\"hum\":65.2}", ts);

    struct mqtt_publish_param pub = {
        .message.topic.topic.utf8 = (uint8_t *)PUB_TOPIC,
        .message.topic.topic.size = strlen(PUB_TOPIC),
        .message.topic.qos = MQTT_QOS_1_AT_LEAST_ONCE,
        .message.payload.data = payload,
        .message.payload.len = strlen(payload),
    };

    mqtt_publish(&client, &pub);
    printk("已发布: %s\n", payload);
}

int main(void)
{
    /* WiFi 连接（通过 net_mgmt 事件等待 DHCP） */
    struct wifi_connect_req_params wifi = {
        .ssid = WIFI_SSID,
        .ssid_length = strlen(WIFI_SSID),
        .psk = WIFI_PSK,
        .psk_length = strlen(WIFI_PSK),
        .security = WIFI_SECURITY_TYPE_PSK,
        .channel = WIFI_CHANNEL_ANY,
    };

    const struct device *net_dev = DEVICE_DT_GET(DT_CHOSEN(zephyr_wifi));
    net_mgmt(NET_REQUEST_WIFI_CONNECT, net_dev,
              &wifi, sizeof(wifi));

    printk("正在连接 WiFi...\n");
    k_sleep(K_SECONDS(5));

    /* MQTT 连接 */
    mqtt_setup();
    k_sleep(K_SECONDS(2));

    /* 订阅控制主题 */
    mqtt_subscribe_cmd();

    /* 主循环：定时发布数据 */
    while (1) {
        mqtt_input(&client);
        mqtt_live(&client);
        publish_sensor_data();
        k_sleep(K_SECONDS(10));
    }

    return 0;
}

低功耗可穿戴设备

可穿戴设备是 Zephyr 最擅长的领域之一。nRF52 系列芯片天生为低功耗 BLE 设计，配合 Zephyr 的电源管理框架，可以实现数月甚至数年的纽扣电池续航。本节我们设计一个心率监测手环的架构方案。

系统架构

可穿戴设备的核心挑战是在极低功耗下完成传感器采集和无线通信。我们的架构设计遵循"快速采集 → 快速发送 → 长时间睡眠"的原则：

  • 心率传感器：通过 I2C 连接 MAX30102 光电脉搏传感器，每次唤醒采集 5 秒心率数据
  • BLE GATT 服务：实现标准 Heart Rate Service (UUID 0x180D)，手机可直接连接读取
  • 电源管理：利用 Zephyr PM 框架，空闲时进入深度睡眠，仅 RTC 定时器保持运行
  • 事件驱动：不使用轮询，所有操作由定时器中断或 BLE 事件触发

关键代码：BLE GATT 心率服务

C 
#include <zephyr/bluetooth/bluetooth.h>
#include <zephyr/bluetooth/gatt.h>
#include <zephyr/bluetooth/uuid.h>
#include <zephyr/pm/pm.h>

#define HR_MEAS_INTERVAL_MS  60000

static uint8_t hr_value = 72;
static bool hr_notify_enabled;

static ssize_t hr_read(struct bt_conn *conn,
                       const struct bt_gatt_attr *attr,
                       void *buf, uint16_t len,
                       uint16_t offset)
{
    return bt_gatt_attr_read(conn, attr, buf, len, offset,
                              &hr_value, sizeof(hr_value));
}

static void hr_ccc_cfg_changed(const struct bt_gatt_attr *attr,
                                   uint16_t value)
{
    hr_notify_enabled = (value == BT_GATT_CCC_NOTIFY);
}

/* Heart Rate Service GATT 定义 */
BT_GATT_SERVICE_DEFINE(hr_svc,
    /* Primary Service: Heart Rate (0x180D) */
    BT_GATT_PRIMARY_SERVICE(BT_UUID_HRS),

    /* Heart Rate Measurement */
    BT_GATT_CHARACTERISTIC(BT_UUID_HRS_MEASUREMENT,
                            BT_GATT_CHRC_READ | BT_GATT_CHRC_NOTIFY,
                            BT_GATT_PERM_READ,
                            hr_read, NULL, NULL),
    BT_GATT_CCC(hr_ccc_cfg_changed,
                BT_GATT_PERM_READ | BT_GATT_PERM_WRITE),

    /* Body Sensor Location */
    BT_GATT_CHARACTERISTIC(BT_UUID_HBSL,
                            BT_GATT_CHRC_READ,
                            BT_GATT_PERM_READ,
                            NULL, NULL, NULL),
);

static void hr_notify(void)
{
    if (!hr_notify_enabled) {
        return;
    }

    uint8_t meas[2] = { 0x00, hr_value };
    bt_gatt_notify(NULL, &hr_svc.attrs[1], meas, sizeof(meas));
}

static void pm_state_exit(void)
{
    /* 唤醒后重新初始化外设 */
    printk("从低功耗状态唤醒\n");
}

int main(void)
{
    bt_enable(NULL);
    printk("可穿戴设备启动\n");

    while (1) {
        /* 1. 唤醒，采集心率 */
        hr_value = read_heart_rate_sensor();
        printk("心率: %d bpm\n", hr_value);

        /* 2. 通过 BLE 通知 */
        hr_notify();

        /* 3. 进入低功耗睡眠 */
        k_msleep(HR_MEAS_INTERVAL_MS);
    }

    return 0;
}
低功耗设计要点
  • 减少唤醒频率：心率不需要每秒监测，每 30~60 秒采集一次即可大幅省电
  • 快速完成工作：唤醒后尽快完成采集和发送，缩短活跃时间
  • 关闭不用的外设：睡眠前通过 pm_device_action_run() 关闭 I2C、ADC 等外设
  • 使用 BLE Connection Parameters：将连接间隔设为较长值（如 1 秒），减少射频唤醒次数
  • 选择合适的 PM 状态：nRF52 支持 System ON Idle、System OFF 等多种状态，根据延迟需求选择

OTA 固件升级实战

OTA（Over-The-Air）固件升级是物联网产品的刚需功能——没有 OTA，你就得派人到现场去刷固件，成本不可接受。Zephyr 通过 MCUboot 引导加载器 + SMP（Simple Management Protocol） 协议来实现安全可靠的 OTA 升级。

OTA 升级原理

MCUboot 采用双 Bank（A/B）方案：Flash 被划分为两个大小相同的 Slot——Slot 0 运行当前固件，Slot 1 存放新固件。升级流程如下：

  1. 通过 SMP 协议将新固件传输到 Slot 1
  2. MCUboot 验证 Slot 1 固件的签名
  3. 重启后 MCUboot 将 Slot 1 标记为待升级
  4. MCUboot 将 Slot 1 的固件交换到 Slot 0（或直接从 Slot 1 启动）
  5. 新固件运行，确认升级成功后标记为"已确认"
  6. 如果新固件运行异常，重启后 MCUboot 自动回滚到旧固件

步骤一：配置 MCUboot

首先在 prj.conf 中启用 MCUboot 和 SMP 服务：

conf 
# 启用 MCUboot
CONFIG_BOOTLOADER_MCUBOOT=y

# 启用 SMP 服务（用于固件传输）
CONFIG_MCUMGR=y
CONFIG_MCUMGR_SMP_BT=y
CONFIG_MCUMGR_SMP_BT_AUTHEN=n
CONFIG_MCUMGR_CMD_IMG_MGMT=y
CONFIG_MCUMGR_CMD_OS_MGMT=y

# BLE 配置
CONFIG_BT=y
CONFIG_BT_PERIPHERAL=y
CONFIG_BT_DEVICE_NAME="OTA-Device"

# Flash 分区
CONFIG_FLASH=y
CONFIG_FLASH_MAP=y
CONFIG_STREAM_FLASH=y

# 图片管理
CONFIG_IMG_MANAGER=y
CONFIG_IMG_ERASE_PROGRESSIVELY=y

步骤二：配置 Flash 分区表

MCUboot 需要知道 Flash 的分区布局。我们在设备树覆盖层中定义分区表：

            label = "image-scratch";
            reg = <0x000DA000 0x0001D000>;

            label = "storage";
            reg = <0x000F7000 0x00008000>;
        };
    };
};

步骤三：应用代码中确认升级

新固件启动后，必须调用 boot_write_img_confirmed() 确认升级成功，否则 MCUboot 会在下次重启时自动回滚：

C 
#include <zephyr/kernel.h>
#include <zephyr/dfu/mcuboot.h>
#include <zephyr/logging/log.h>

LOG_MODULE_REGISTER(main, LOG_LEVEL_INF);

int main(void)
{
    bool is_upgrade = mcuboot_swap_type() == BOOT_SWAP_TYPE_REVERT;

    if (is_upgrade) {
        LOG_INF("检测到 OTA 升级，运行自检...");

        /* 执行自检，确认新固件运行正常 */
        int ret = self_test();
        if (ret == 0) {
            boot_write_img_confirmed();
            LOG_INF("OTA 升级已确认，回滚保护生效");
        } else {
            LOG_ERR("自检失败！重启后将自动回滚");
        }
    }

    /* 正常应用逻辑 */
    while (1) {
        k_sleep(K_SECONDS(1));
    }

    return 0;
}

步骤四：使用 mcumgr 推送固件

在电脑端安装 mcumgr 命令行工具，通过 BLE 将新固件推送到设备：

bash 
# 安装 mcumgr
go install github.com/apache/mynewt-mcumgr/cli/mcumgr@latest

# 连接设备（通过 BLE）
mcumgr --conntype ble --connstring peer_name=OTA-Device \
    image list

# 上传新固件
mcumgr --conntype ble --connstring peer_name=OTA-Device \
    image upload build/zephyr/app_update.bin

# 标记新固件为待升级
mcumgr --conntype ble --connstring peer_name=OTA-Device \
    image test <hash-of-new-image>

# 重启设备，MCUboot 自动完成升级
mcumgr --conntype ble --connstring peer_name=OTA-Device \
    reset

重要：上传固件时使用的是 app_update.bin 而不是 zephyr.bin。前者是经过 MCUboot 签名处理的固件包，后者是原始的裸二进制文件。构建时需要使用 west build -t signed_bin 或确保构建系统自动生成签名后的固件。

安全警告：生产环境中必须启用固件签名验证（CONFIG_MCUBOOT_SIGNATURE_KEY_FILE），否则任何人都可以推送恶意固件。MCUboot 支持 RSA 和 ECDSA 签名算法，推荐使用 ECDSA-P256 以减少签名开销。

自定义开发板移植

当你的项目使用了一块 Zephyr 官方未支持的开发板时，就需要自己动手移植（Porting）。移植的本质就是告诉 Zephyr："这块板子用了什么芯片、引脚怎么连、有哪些外设"。听起来复杂，但只要按步骤来，其实并不难。

移植步骤总览

  1. 创建板级目录：在正确位置建立板级文件结构
  2. 编写 board.yml：告诉构建系统这块板的基本信息
  3. 创建 Pinmux 配置：定义引脚复用功能
  4. 编写设备树文件：描述板上的硬件连接
  5. 添加 Kconfig：提供板级配置选项
  6. 测试与调试：编译、烧录、验证

步骤一：创建板级目录

从 Zephyr 3.5 开始，自定义板放在应用的 boards/ 目录下。目录命名规则为 <vendor>/<board_name>：

text 
my-app/
├── boards/
│   └── mycompany/
│       └── my_board/
│           ├── board.yml            # 板级描述文件

步骤二：编写 board.yml

board.yml 是板级的"身份证"，告诉 Zephyr 这块板的基本信息：

yaml 
board:
  name: my_board
  full_name: My Company Custom Board
  vendor: mycompany
  socs:
    - name: stm32f407xx

步骤三：编写设备树文件

设备树是移植的核心。你需要描述板上的所有硬件连接——哪些引脚连了 LED、按钮、串口、I2C 总线等。通常先包含芯片级的设备树，然后在板级文件中补充板特有的配置：

dts 
/* my_board.dts */
/dts-v1/;

#include <st/f4/stm32f407Zg.dtsi>
#include <st/f4/stm32f407Zg-pinctrl.dtsi>
#include "my_board-pinctrl.dtsi"

/ {
    model = "My Company Custom Board";
    compatible = "mycompany,my-board";

    chosen {
        zephyr,console = &usart3;
        zephyr,shell-uart = &usart3;
        zephyr,sram = &sram0;
        zephyr,flash = &flash0;
        zephyr,ccm = &ccm0;
    };

    leds {
        compatible = "gpio-leds";
        led0: led_0 {
            gpios = &gpiod, 12, GPIO_ACTIVE_LOW;
            label = "Status LED";
        };
        led1: led_1 {
            gpios = &gpiod, 13, GPIO_ACTIVE_LOW;
            label = "Error LED";
        };
    };

    buttons {
        compatible = "gpio-keys";
        btn0: button_0 {
            gpios = &gpioc, 0, GPIO_ACTIVE_LOW;
            label = "User Button";
        };
    };

    aliases {
        led0 = &led0;
        led1 = &led1;
        sw0 = &btn0;
    };
};

/* 启用板上的外设 */
&usart3 {
    pinctrl-0 = <&usart3_default>;
    pinctrl-names = "default";
    current-speed = <115200>;
    status = "okay";
};

&i2c1 {
    pinctrl-0 = <&i2c1_default>;
    pinctrl-names = "default";
    status = "okay";
    clock-frequency = <I2C_BITRATE_FAST>;
};

步骤四：编写引脚控制文件

引脚控制文件定义了每个外设使用的具体 GPIO 引脚和复用功能：

dts 
/* my_board-pinctrl.dtsi */
&pinctrl {
    usart3_default: usart3_default {
        group1 {
            pinmux = <STM32F4_PINMUX_FUNC_PD8_USART3_TX>;
            bias-disable;
            drive-push-pull;
        };
        group2 {
            pinmux = <STM32F4_PINMUX_FUNC_PD9_USART3_RX>;
            bias-pull-up;
        };
    };

    i2c1_default: i2c1_default {
        group1 {
            pinmux = <STM32F4_PINMUX_FUNC_PB6_I2C1_SCL,
                     STM32F4_PINMUX_FUNC_PB7_I2C1_SDA>;
            bias-pull-up;
            drive-open-drain;
        };
    };
};

步骤六：编译与测试

bash 
# 编译（指定自定义板）
west build -b mycompany/my_board -p

# 烧录
west flash

# 打开串口终端观察输出
west esptool --port COM3 monitor

调试技巧：如果板子没有任何输出，首先检查 chosen 节点中 zephyr,console 指向的串口是否正确，然后确认 pinctrl 中 TX/RX 引脚是否和原理图一致。90% 的移植问题都是引脚配置错误导致的。

快速移植捷径：如果你的板子和某个官方板使用相同的 SoC，最快的办法是复制官方板的目录，然后修改设备树中的引脚配置和外设差异。比如你的板子用的是 STM32F407，可以复制 nucleo_f429zi 的板级文件作为起点，只改不同的部分。站在巨人的肩膀上，事半功倍！

移植检查清单
  • ✅ board.yml 中的 SoC 名称与 Zephyr 支持的 SoC 匹配
  • ✅ 设备树 chosen 节点正确指定了 console、sram、flash
  • ✅ pinctrl 中所有引脚与原理图一致
  • ✅ LED 和按钮的 GPIO 编号正确
  • ✅ 时钟配置（HSE/HSI 频率、PLL 倍频系数）与硬件匹配
  • ✅ 编译无错误，烧录后串口有输出
  • ✅ GPIO、UART、I2C 等基本外设功能验证通过
