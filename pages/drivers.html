设备驱动模型

深入理解 Zephyr RTOS 的设备驱动架构 —— 从硬件描述到驱动实现，从通用 API 到自定义驱动开发

驱动模型架构

Zephyr 的驱动模型是整个系统最精妙的设计之一。它的核心思想是解耦——让硬件描述、驱动实现和应用程序三者各司其职，互不干扰。

打个比方：驱动 API 就像万能遥控器。不管你家是格力空调还是美的空调，遥控器上的"开机""调温"按钮都是一样的。你不需要知道空调内部怎么工作，只需要按按钮就行。Zephyr 的通用 API 就是这个万能遥控器，驱动实现就是各家空调的接收器，设备树就是空调的品牌型号说明书。

三层架构

Zephyr 驱动模型采用清晰的三层架构：

  • 应用层：调用通用 API（如 gpio_pin_set()、i2c_write()），不关心底层硬件细节
  • 通用 API 层：定义统一的函数接口（struct gpio_driver_api 等），是应用和驱动之间的桥梁
  • 驱动实现层：针对具体硬件实现 API，操作寄存器、配置时钟等
设备树 → 驱动 → 应用的数据流

设备树（Device Tree）描述板子上有哪些硬件、地址在哪、用什么引脚 → 驱动根据设备树信息初始化硬件并注册到系统 → 应用通过通用 API 访问设备，完全不需要知道硬件细节。这种分层设计让同一份应用代码可以在不同硬件平台上运行，只需更换设备树和对应的驱动即可。

DEVICE_DEFINE 与驱动注册

每个驱动通过 DEVICE_DEFINE 宏将自身注册到系统中。这个宏在编译期就完成了驱动的"登记"工作——包括初始化函数、驱动 API 结构体、设备数据等。系统启动时，按优先级自动调用各驱动的 init 函数。

C 
/* 驱动 API 结构体：定义该类设备的统一接口 */
struct my_driver_api {
    int (*open)(const struct device *dev);
    int (*read)(const struct device *dev, uint8_t *buf, size_t len);
    int (*write)(const struct device *dev, const uint8_t *buf, size_t len);
    void (*close)(const struct device *dev);
};

/* 驱动初始化函数 */
static int my_driver_init(const struct device *dev)
{
    /* 硬件初始化：配置寄存器、使能时钟等 */
    return 0;
}

/* 实例化 API 结构体 */
static const struct my_driver_api my_api = {
    .open  = my_driver_open,
    .read  = my_driver_read,
    .write = my_driver_write,
    .close = my_driver_close,
};

/* 注册驱动到系统 */
DEVICE_DEFINE(my_device,            /* 设备实例名 */
               "MY_DEVICE",          /* 设备名称字符串 */
               &my_driver_init,      /* 初始化函数 */
               NULL,                  /* pm_device（电源管理） */
               &my_data,              /* 设备私有数据 */
               &my_config,            /* 设备配置信息 */
               POST_KERNEL,          /* 初始化优先级等级 */
               50,                    /* 初始化优先级数值 */
               &my_api);              /* API 结构体指针 */

小贴士：初始化优先级等级分为 EARLY、PRE_KERNEL_1、PRE_KERNEL_2、POST_KERNEL、APPLICATION。大多数驱动使用 POST_KERNEL，只有极少数必须在调度器启动前就绪的驱动（如中断控制器）才用更早的等级。

GPIO 输入输出

GPIO（General Purpose Input/Output）是最基础的外设接口，就像电灯开关——你可以控制开关的开与关（输出高低电平），也可以检测开关是否被按下（读取输入电平）。Zephyr 的 GPIO 驱动提供了引脚配置、输入读取、输出控制和中断处理四大核心功能。

核心 API

  • gpio_pin_configure() — 配置引脚方向和属性（输入/输出、上拉/下拉等）
  • gpio_port_set_bits_raw() — 将指定引脚设为高电平（原子操作）
  • gpio_port_clear_bits_raw() — 将指定引脚设为低电平
  • gpio_pin_get() — 读取引脚当前电平
  • gpio_pin_interrupt_configure() — 配置引脚中断触发方式

示例：LED 闪烁 + 按键中断

C 
#include <zephyr/kernel.h>
#include <zephyr/drivers/gpio.h>

#define LED_NODE   DT_NODELABEL(led0)
#define BUTTON_NODE DT_NODELABEL(button0)

static const struct gpio_dt_spec led = GPIO_DT_SPEC_GET(LED_NODE, gpios);
static const struct gpio_dt_spec btn = GPIO_DT_SPEC_GET(BUTTON_NODE, gpios);
static struct gpio_callback btn_cb_data;

static void button_pressed(const struct device *dev,
                          struct gpio_callback *cb,
                          uint32_t pins)
{
    printk("按键被按下！引脚掩码: 0x%x\n", pins);
    gpio_pin_toggle_dt(&led);
}

int main(void)
{
    /* 配置 LED 引脚为输出，初始低电平 */
    gpio_pin_configure_dt(&led, GPIO_OUTPUT_ACTIVE);

    /* 配置按键引脚为输入 */
    gpio_pin_configure_dt(&btn, GPIO_INPUT);

    /* 配置中断：下降沿触发（按下时） */
    gpio_pin_interrupt_configure_dt(&btn, GPIO_INT_EDGE_TO_ACTIVE);

    /* 注册回调 */
    gpio_init_callback(&btn_cb_data, button_pressed, BIT(btn.pin));
    gpio_add_callback(btn.port, &btn_cb_data);

    /* LED 闪烁主循环 */
    while (1) {
        gpio_pin_toggle_dt(&led);
        k_msleep(500);
    }

    return 0;
}

注意：中断回调函数运行在中断上下文中，不能执行阻塞操作（如 k_msleep()、k_sem_take()）。如果需要在中断中做耗时操作，应该使用工作队列（k_work）将处理推迟到线程上下文。

I2C 通信

I2C 是一种两线制串行总线，就像一条公交路线——一条线上挂着多个站点（从设备），每个站点有唯一的地址。主设备是公交车，按地址找到对应的站点进行数据交换。SCL 是时钟线（公交车的时刻表），SDA 是数据线（乘客上下车）。

核心 API

  • i2c_write() — 向从设备写入数据
  • i2c_read() — 从从设备读取数据
  • i2c_write_read() — 先写后读（最常见的"写寄存器地址 → 读数据"模式）
  • i2c_transfer() — 底层传输，使用 i2c_msg 结构体，支持连续多次读写

示例：通过 I2C 读取温度传感器

C 
#include <zephyr/kernel.h>
#include <zephyr/drivers/i2c.h>

/* 假设使用 TMP102 温度传感器，7位地址 0x48 */
#define TEMP_SENSOR_ADDR 0x48
#define TEMP_REG          0x00

int read_temperature(const struct device *i2c_dev)
{
    uint8_t reg = TEMP_REG;
    uint8_t buf[2];
    int ret;

    /* 方式一：i2c_write_read — 最简洁的写后读 */
    ret = i2c_write_read(i2c_dev, TEMP_SENSOR_ADDR,
                        &reg, 1,    /* 写：寄存器地址 */
                        buf, 2);      /* 读：2字节温度数据 */
    if (ret < 0) {
        printk("I2C 读取失败: %d\n", ret);
        return ret;
    }

    /* TMP102: 12位温度值，左对齐在两个字节中 */
    int16_t raw = (buf[0] << 4) | (buf[1] >> 4);
    if (raw & 0x800) {
        raw |= 0xF000;  /* 负温度：符号扩展 */
    }
    float temp_c = raw * 0.0625f;
    printk("温度: %.2f °C\n", temp_c);

    return 0;
}

int main(void)
{
    const struct device *i2c = DEVICE_DT_GET(DT_NODELABEL(i2c0));

    if (!device_is_ready(i2c)) {
        printk("I2C 设备未就绪\n");
        return -1;
    }

    while (1) {
        read_temperature(i2c);
        k_msleep(1000);
    }

    return 0;
}

小贴士：i2c_write_read() 内部会自动处理重复起始条件（Repeated START），确保读操作不会被其他主设备插队。如果用 i2c_write() + i2c_read() 分开调用，中间总线可能被释放，在多主环境中可能出现问题。

SPI 通信

SPI 是一种高速同步串行总线，就像一条四车道高速公路——MOSI 是去程车道（主发从收），MISO 是回程车道（从发主收），SCLK 是交通信号灯（时钟同步），CS 是收费站（选中哪个从设备）。SPI 是全双工通信，收发同时进行，速度远超 I2C。

SPI 四种工作模式

SPI 的工作模式由时钟极性（CPOL）和时钟相位（CPHA）决定：

  • Mode 0（CPOL=0, CPHA=0）：空闲低电平，第一个边沿采样——最常用
  • Mode 1（CPOL=0, CPHA=1）：空闲低电平，第二个边沿采样
  • Mode 2（CPOL=1, CPHA=0）：空闲高电平，第一个边沿采样
  • Mode 3（CPOL=1, CPHA=1）：空闲高电平，第二个边沿采样

示例：SPI 数据传输

C 
#include <zephyr/kernel.h>
#include <zephyr/drivers/spi.h>

#define SPI_NODE DT_NODELABEL(spi1)

static const struct spi_dt_spec spi_spec = SPI_DT_SPEC_GET(
    SPI_NODE,
    SPI_OP_MODE_MASTER | SPI_MODE_CPOL | SPI_MODE_CPHA
    | SPI_WORD_SET(8) | SPI_LINES_SINGLE,
    0   /* CS 延迟（微秒） */
);

int spi_transaction(void)
{
    uint8_t tx_buf[] = { 0x9F };          /* 命令：读取 JEDEC ID */
    uint8_t rx_buf[4] = { 0 };             /* 接收缓冲区 */
    struct spi_buf tx = { .buf = tx_buf, .len = sizeof(tx_buf) };
    struct spi_buf rx = { .buf = rx_buf, .len = sizeof(rx_buf) };
    struct spi_buf_set tx_set = { .buffers = &tx, .count = 1 };
    struct spi_buf_set rx_set = { .buffers = &rx, .count = 1 };

    /* 全双工收发 */
    int ret = spi_transceive_dt(&spi_spec, &tx_set, &rx_set);
    if (ret < 0) {
        printk("SPI 传输失败: %d\n", ret);
        return ret;
    }

    printk("JEDEC ID: %02x %02x %02x\n", rx_buf[1], rx_buf[2], rx_buf[3]);
    return 0;
}

int main(void)
{
    if (!spi_is_ready_dt(&spi_spec)) {
        printk("SPI 设备未就绪\n");
        return -1;
    }

    spi_transaction();
    return 0;
}

注意：SPI 的 CS 片选信号可以在设备树中指定 GPIO，Zephyr 会自动在传输前拉低 CS、传输后拉高 CS。如果需要手动控制 CS，可以在 spi_config 中设置 SPI_HOLD_ON_CS 标志。

UART 串口通信

UART 是最经典的串行通信方式，就像两个人用对讲机通话——一个说一个听，按固定速率（波特率）收发数据。Zephyr 提供了三种 UART 使用模式：轮询（Poll）、中断（Interrupt）和异步（Async）。

三种模式对比

模式 API 特点 适用场景
轮询 uart_poll_in/out CPU 空转等待，最简单 调试输出、简单通信
中断 uart_irq_callback_set 数据到达时触发中断，CPU 利用率高 大多数实际应用
异步 uart_tx/rx_enable DMA 传输，零 CPU 开销 高速通信、低功耗场景

示例：中断驱动接收

C 
#include <zephyr/kernel.h>
#include <zephyr/drivers/uart.h>

#define UART_NODE DT_NODELABEL(uart0)

static const struct device *uart_dev;
static uint8_t rx_buf[64];
static size_t rx_idx = 0;

static void uart_irq_callback(const struct device *dev, void *user_data)
{
    uint8_t c;

    while (uart_irq_update(dev) && uart_irq_rx_ready(dev)) {
        while (uart_fifo_read(dev, &c, 1) == 1) {
            if (c == '\n' || rx_idx >= sizeof(rx_buf) - 1) {
                rx_buf[rx_idx] = '\0';
                printk("收到: %s\n", rx_buf);
                rx_idx = 0;
            } else {
                rx_buf[rx_idx++] = c;
            }
        }
    }
}

int main(void)
{
    uart_dev = DEVICE_DT_GET(UART_NODE);
    if (!device_is_ready(uart_dev)) {
        printk("UART 设备未就绪\n");
        return -1;
    }

    uart_irq_callback_set(uart_dev, uart_irq_callback, NULL);
    uart_irq_rx_enable(uart_dev);

    while (1) {
        k_msleep(1000);
    }

    return 0;
}

小贴士：在中断回调中，务必先调用 uart_irq_update() 确认中断状态已更新，再用 uart_irq_rx_ready() 检查是否有数据。循环读取 FIFO 直到读空，否则中断可能不会再次触发。

PWM 脉冲宽度调制

PWM 就像水龙头的开关节奏——你快速地开关水龙头，开的时间长，接到的水就多（占空比高）；开的时间短，接到的水就少（占空比低）。虽然水龙头只有开和关两个状态，但通过控制开关的比例，就能"模拟"出不同的水流大小。这就是 PWM 控制亮度、速度的原理。

核心概念

  • 周期（Period）：一个完整 PWM 循环的时间
  • 脉冲宽度（Pulse）：高电平持续的时间
  • 占空比（Duty Cycle）：脉冲宽度 / 周期 × 100%

示例：控制 LED 亮度渐变

C 
#include <zephyr/kernel.h>
#include <zephyr/drivers/pwm.h>

#define PWM_NODE DT_NODELABEL(pwm_led0)

static const struct pwm_dt_spec pwm_led = PWM_DT_SPEC_GET(PWM_NODE);

#define PERIOD_USEC  20000   /* 周期 20ms = 50Hz */
#define STEP_USEC     200     /* 每步变化 200μs */

int main(void)
{
    if (!device_is_ready(pwm_led.dev)) {
        printk("PWM 设备未就绪\n");
        return -1;
    }

    uint32_t pulse = 0;
    int direction = 1;

    while (1) {
        /* 设置 PWM：周期 20ms，脉冲宽度 pulse μs */
        pwm_set_usec(pwm_led.dev, pwm_led.channel,
                     PERIOD_USEC, pulse);

        /* 呼吸灯效果：亮度渐增再渐减 */
        pulse += direction * STEP_USEC;
        if (pulse >= PERIOD_USEC) {
            pulse = PERIOD_USEC;
            direction = -1;
        } else if (pulse == 0) {
            direction = 1;
        }

        k_msleep(20);
    }

    return 0;
}

控制舵机时，通常使用 20ms 周期（50Hz），脉冲宽度 0.5ms~2.5ms 对应 0°~180° 转角。用 pwm_set_usec() 可以很方便地设置微秒级的脉冲宽度。

ADC 模数转换

ADC 就像翻译官——它把模拟世界的连续信号（电压）翻译成数字世界的离散数值，让 MCU 能理解和处理。比如温度传感器输出一个随温度变化的电压，ADC 把这个电压转换成数字值，程序就能计算出实际温度。

核心概念

  • 分辨率（Resolution）：ADC 的"刻度精度"，如 12 位 ADC 将量程分成 2¹² = 4096 级
  • 参考电压（Reference）：ADC 的满量程电压基准
  • 增益（Gain）：输入信号的放大倍数，可测量更小的电压范围

示例：读取模拟电压

C 
#include <zephyr/kernel.h>
#include <zephyr/drivers/adc.h>

#define ADC_NODE       DT_NODELABEL(adc0)
#define ADC_CHANNEL    0
#define ADC_RESOLUTION 12

static const struct device *adc_dev;

static struct adc_channel_cfg channel_cfg = {
    .gain             = ADC_GAIN_1,         /* 增益 1x */
    .reference        = ADC_REF_INTERNAL,    /* 内部参考电压 */
    .acquisition_time = ADC_ACQ_TIME_DEFAULT,
    .channel_id       = ADC_CHANNEL,
    .differential     = 0,                   /* 单端输入 */
};

static int16_t sample_buffer[1];

static struct adc_sequence sequence = {
    .channels    = BIT(ADC_CHANNEL),
    .buffer      = sample_buffer,
    .buffer_size = sizeof(sample_buffer),
    .resolution  = ADC_RESOLUTION,
};

int main(void)
{
    adc_dev = DEVICE_DT_GET(ADC_NODE);
    if (!device_is_ready(adc_dev)) {
        printk("ADC 设备未就绪\n");
        return -1;
    }

    adc_channel_setup(adc_dev, &channel_cfg);

    while (1) {
        int ret = adc_read(adc_dev, &sequence);
        if (ret == 0) {
            /* 将原始值转换为电压（mV） */
            int32_t raw = sample_buffer[0];
            adc_raw_to_millivolts(
                adc_ref_internal(adc_dev),
                channel_cfg.gain,
                ADC_RESOLUTION,
                &raw);
            printk("ADC 原始值: %d, 电压: %d mV\n",
                   sample_buffer[0], raw);
        }
        k_msleep(500);
    }

    return 0;
}

注意：adc_raw_to_millivolts() 会就地修改传入的值，所以如果还需要原始值，请提前备份。另外，不同 MCU 的内部参考电压不同（常见的有 1.8V、2.5V、3.3V），请查阅数据手册确认。

DMA 直接内存访问

DMA 就像快递员——你把要搬运的包裹（数据）和地址告诉快递员，快递员自己去搬运，你不用亲力亲为。CPU 只需要发起一次 DMA 传输请求，之后就可以去做别的事情，DMA 控制器会在后台自动完成数据搬运，搬运完了通知 CPU。

DMA 的两种典型场景

  • 内存到内存（Memory-to-Memory）：快速搬运大块数据，比如图像缓冲区拷贝。速度远超 CPU 逐字节复制。
  • 外设到内存（Peripheral DMA）：UART、SPI、I2C 等外设收发数据时，DMA 自动在外设寄存器和内存缓冲区之间搬运数据，CPU 几乎零开销。这就是 UART 异步模式高效的秘密。

DMA 通道与配置

每个 DMA 控制器有多个通道，每个通道可以独立配置：

  • 源地址和目标地址：数据从哪来、到哪去
  • 传输宽度：每次搬运 1/2/4 字节
  • 块大小：一次 DMA 请求搬运的数据量
  • 回调函数：传输完成时的通知

DMA 配置与具体硬件强相关，不同 MCU 的 DMA 控制器差异较大。使用 DMA 时务必参考芯片数据手册中的 DMA 通道映射表，确保外设请求信号连接到正确的 DMA 通道。配置错误可能导致数据丢失或系统崩溃。

传感器驱动

Zephyr 的传感器子系统就像统一的仪表盘——不管你用的是加速度计、陀螺仪、温度计还是湿度计，读取数据的方式都一样：先 sensor_sample_fetch() 采样，再 sensor_channel_get() 取值。这种统一的接口让你可以轻松替换传感器硬件，而应用代码几乎不用改。

核心 API

  • sensor_sample_fetch() — 从传感器获取一次采样（触发硬件测量）
  • sensor_channel_get() — 读取指定通道的采样值
  • sensor_sample_fetch_chan() — 只获取指定通道的采样

sensor_value 结构体

传感器返回的值用 struct sensor_value 表示，包含两个 int32_t 字段：val1 是整数部分，val2 是小数部分（以百万分之一为单位）。实际值 = val1 + val2 × 10⁻⁶。

示例：读取加速度计数据

C 
#include <zephyr/kernel.h>
#include <zephyr/drivers/sensor.h>

/* 从设备树获取加速度计节点 */
static const struct device *accel = DEVICE_DT_GET(DT_NODELABEL(accel0));

static void print_sensor_value(const struct sensor_value *val)
{
    printk("%d.%06d", val->val1, abs(val->val2));
}

int main(void)
{
    if (!device_is_ready(accel)) {
        printk("加速度计未就绪\n");
        return -1;
    }

    while (1) {
        /* 第一步：触发采样 */
        int ret = sensor_sample_fetch(accel);
        if (ret < 0) {
            printk("采样失败: %d\n", ret);
            k_msleep(100);
            continue;
        }

        /* 第二步：读取各轴数据 */
        struct sensor_value accel_x, accel_y, accel_z;

        sensor_channel_get(accel, SENSOR_CHAN_ACCEL_X, &accel_x);
        sensor_channel_get(accel, SENSOR_CHAN_ACCEL_Y, &accel_y);
        sensor_channel_get(accel, SENSOR_CHAN_ACCEL_Z, &accel_z);

        printk("加速度 (m/s²): X=");
        print_sensor_value(&accel_x);
        printk(" Y=");
        print_sensor_value(&accel_y);
        printk(" Z=");
        print_sensor_value(&accel_z);
        printk("\n");

        k_msleep(100);
    }

    return 0;
}
Zephyr 支持的传感器类型

Zephyr 内置了大量传感器驱动，包括但不限于：BME280（温湿度气压）、MPU6050（六轴 IMU）、LIS2DH（加速度计）、BMM150（磁力计）、VL53L0X（激光测距）、SHT3x（温湿度）等。完整列表参见 Zephyr 官方文档的 Sensor Drivers 章节。

自定义驱动开发

当 Zephyr 内置驱动无法满足需求时——比如你用了一颗冷门芯片，或者需要驱动自定义的 FPGA 外设——就需要自己写驱动了。别担心，Zephyr 的驱动模型非常规范，按照以下步骤一步步来就行。

开发步骤

  1. 定义驱动 API 结构体：描述你的设备提供哪些操作
  2. 实现初始化函数：硬件复位、寄存器配置、中断设置
  3. 使用 DEVICE_DEFINE 注册驱动：将驱动纳入系统管理
  4. 添加 Kconfig 选项：让用户可以在 menuconfig 中启用/禁用你的驱动
  5. 添加设备树绑定：让设备树能描述你的硬件信息

最小自定义驱动骨架

C 
/* ===== my_driver.h — 对外头文件 ===== */
#include <zephyr/device.h>

/* 定义驱动 API 结构体 */
struct my_driver_api {
    int (*do_something)(const struct device *dev, int param);
    int (*get_status)(const struct device *dev);
};

/* 便捷包装函数，让用户不用直接操作 API 结构体 */
static inline int my_driver_do_something(const struct device *dev, int param)
{
    const struct my_driver_api *api =
        (const struct my_driver_api *)dev->api;
    return api->do_something(dev, param);
}

/* ===== my_driver.c — 驱动实现 ===== */
#include "my_driver.h"

/* 设备私有数据 */
struct my_driver_data {
    int state;
    uint8_t buffer[32];
};

/* 设备配置信息（来自设备树） */
struct my_driver_config {
    uint32_t base_addr;
    int irq_num;
};

static int my_do_something(const struct device *dev, int param)
{
    struct my_driver_data *data = dev->data;
    data->state = param;
    return 0;
}

static int my_get_status(const struct device *dev)
{
    struct my_driver_data *data = dev->data;
    return data->state;
}

static int my_driver_init(const struct device *dev)
{
    struct my_driver_data *data = dev->data;
    data->state = 0;
    printk("自定义驱动初始化完成\n");
    return 0;
}

static const struct my_driver_api my_api = {
    .do_something = my_do_something,
    .get_status   = my_get_status,
};

static struct my_driver_data my_data;
static const struct my_driver_config my_config = {
    .base_addr = DT_REG_ADDR(DT_NODELABEL(mydev)),
    .irq_num   = DT_IRQN(DT_NODELABEL(mydev)),
};

DEVICE_DEFINE(my_driver, "MY_DRIVER",
               my_driver_init, NULL,
               &my_data, &my_config,
               POST_KERNEL, 50,
               &my_api);

Kconfig 配置

在驱动的 Kconfig 文件中添加配置选项：

Kconfig 
config MY_DRIVER
    bool "My custom driver"
    default y
    depends on DT_HAS_MYVENDOR_MYDRIVER_ENABLED
    help
      Enable the custom driver for MyDevice.

config MY_DRIVER_INIT_PRIORITY
    int "My driver init priority"
    default 50
    depends on MY_DRIVER
    help
      Initialization priority for the custom driver.

设备树绑定

创建 YAML 绑定文件，让设备树能描述你的硬件：

YAML 
# dts/bindings/myvendor,mydriver.yaml
description: My custom device driver

compatible: "myvendor,mydriver"

include: [base.yaml]

properties:
  reg:
    required: true
  interrupts:
    required: true

然后在板级设备树中使用：

DTS 
/* 在板级设备树中添加自定义设备节点 */
mydev: mydev@40001000 {
    compatible = "myvendor,mydriver";
    reg = <0x40001000 0x100>;
    interrupts = <5 0>;
};

小贴士：自定义驱动推荐放在项目的 drivers/ 目录下，并在 CMakeLists.txt 中通过 zephyr_include_directories() 和 target_sources() 添加到构建系统。如果驱动比较通用，也可以考虑提交到 Zephyr 上游仓库，让更多人受益。

自定义驱动开发中最常见的陷阱：忘记检查 device_is_ready()。在访问设备之前，务必调用此函数确认驱动初始化成功。否则，如果硬件初始化失败，后续操作可能导致未定义行为甚至系统崩溃。
